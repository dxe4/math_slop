const x=1,y=0,z=1,delta=0.1;
const report = (value) => {
  const check = (v) => {
    if (typeof v === 'number' && !Number.isFinite(v))
      throw Error(
        'Numeric output is outside binary64 representability; use symbolic mathematics or representable inputs.',
      );
    if (v && typeof v === 'object') Object.values(v).forEach(check);
  };
  check(value);
  console.log(JSON.stringify(value));
};
if (
  ![x, y, z].every(Number.isFinite) ||
  Math.max(Math.abs(x), Math.abs(y), Math.abs(z)) > 5
)
  throw Error('Require finite world coordinates within ±5 m.');
const F = (u, v, w) => u * u + v * v - w * w,
  maxComponent = Math.max(Math.abs(x), Math.abs(y), Math.abs(z));
const scaled =
    maxComponent === 0
      ? [0, 0, 0]
      : [x / maxComponent, y / maxComponent, z / maxComponent],
  norm = Math.hypot(...scaled);
if (norm !== 0 && Math.abs(F(...scaled.map((v) => v / norm))) > 1e-12)
  throw Error('Point is off F=0; choose a point on the cone.');
const gradient = [2 * x, 2 * y, -2 * z],
  radius = Math.hypot(scaled[0], scaled[1]);
if (norm === 0) {
  report({
    experiment: 'level-tangents',
    units: 'm',
    input: { x, y, z },
    result: {
      status: 'singular—no unique tangent plane for this cone',
      generators: [
        [1, 0, 1],
        [0, 1, 1],
        [-1, 0, 1],
      ],
    },
  });
} else {
  const angular = [-scaled[1] / radius, scaled[0] / radius, 0],
    generator = scaled.map((v) => v / norm),
    probe = (v) => F(x + delta * v[0], y + delta * v[1], z + delta * v[2]);
  report({
    experiment: 'level-tangents',
    units: { gradient: 'm', residual: 'm²' },
    input: { x, y, z },
    result: {
      gradient,
      tangentEquation: '∇F(P)·(X−P)=0',
      angular,
      generator,
      gradientDots: [
        gradient.reduce((s, g, i) => s + g * angular[i], 0),
        gradient.reduce((s, g, i) => s + g * generator[i], 0),
      ],
      angularResidual: probe(angular),
      generatorResidual: probe(generator),
    },
  });
}

Analytic comparison and limits

For F=x2+y2−z2=0F=x^2+y^2-z^2=0, at a regular point the tangent plane has normal ∇F=(2x,2y,−2z)\nabla F=(2x,2y,-2z). Angular direction yields F(P+δv)=δ2F(P+\delta v)=\delta^2 with δ=0.1 m\delta=0.1\,\mathrm m; generator direction yields zero. At the apex this cone has no unique tangent plane, but zero gradient does not imply this for arbitrary level sets. Normalize by the largest coordinate before forming unit directions and the relative level check. This avoids an absolute tolerance floor and tiny-norm underflow. Numerical membership tolerance does not prove an exact cone identity. Non-finite numeric output raises an explicit precision error; null is reserved for a deliberately absent derivative.

Boundary and failure behavior
