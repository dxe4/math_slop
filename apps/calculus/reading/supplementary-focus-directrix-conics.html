const e=2,d=3;
const report = (value) => {
  const check = (v) => {
    if (typeof v === 'number' && !Number.isFinite(v))
      throw Error(
        'Numeric output is outside binary64 representability; use symbolic mathematics or representable inputs.',
      );
    if (v && typeof v === 'object') Object.values(v).forEach(check);
  };
  check(value);
  console.log(JSON.stringify(value));
};
if (
  !Number.isFinite(e) ||
  !Number.isFinite(d) ||
  e <= 0 ||
  e > 3 ||
  d <= 0 ||
  d > 10
)
  throw Error('Require 0<e≤3 and 0<d≤10 m.');
const leftLimit = e > 1 ? Math.acos(-1 / e) : Math.PI,
  vertices = (
    e < 1
      ? [(e * d) / (1 + e), (-e * d) / (1 - e)]
      : [(e * d) / (1 + e), ...(e > 1 ? [(e * d) / (e - 1)] : [])]
  ).sort((u, v) => u - v);
const sample = (limit, denominator) =>
  Array.from({ length: 121 }, (_, i) => {
    const angle = (-0.98 + (1.96 * i) / 120) * limit,
      r = (e * d) / denominator(angle);
    return [r * Math.cos(angle), r * Math.sin(angle)];
  });
const left = sample(leftLimit, (a) => 1 + e * Math.cos(a));
plot(left, 'Left positive-radius branch: world x,y (m)');
let right = [];
if (e > 1) {
  right = sample(Math.acos(1 / e), (a) => e * Math.cos(a) - 1);
  plot(right, 'Right positive-radius branch: world x,y (m)');
}
const ratio = ([x, y]) => Math.hypot(x, y) / Math.abs(x - d);
report({
  experiment: 'focus-directrix-conics',
  units: 'm',
  input: { e, d },
  result: {
    kind: e < 1 ? 'ellipse' : e === 1 ? 'parabola' : 'hyperbola',
    vertices,
    leftLimit,
    rightLimit: e > 1 ? Math.acos(1 / e) : null,
    focusDirectrixRatios: [
      ratio(left[60]),
      ...(right.length ? [ratio(right[60])] : []),
    ],
  },
});

Analytic comparison and limits

x2+y2=e∣x−d∣\sqrt{x^2+y^2}=e|x-d|. For e>1e>1 left R=ed1+ecos⁡θR=\frac{ed}{1+e\cos\theta}, ∣θ∣<arccos⁡(−1/e)|\theta|<\arccos(-1/e); right R=edecos⁡ϕ−1R=\frac{ed}{e\cos\phi-1}, ∣ϕ∣<arccos⁡(1/e)|\phi|<\arccos(1/e). Vertices ed1+e\frac{ed}{1+e} and ede−1\frac{ed}{e-1}. Denominator limits have no finite point; clipped samples cannot establish the global locus. Non-finite numeric output raises an explicit precision error; null is reserved for a deliberately absent derivative.

Default and changed-input cases

  • Default e=2,d=3 me=2,d=3\,\mathrm m: left and right vertices x=2 and 6 mx=2\text{ and }6\,\mathrm m, center x=4x=4 m\mathrm m, left angular limit 2π/32\pi/3, right limit π/3\pi/3; each sampled focus/directrix ratio is 2.
  • Changed e=0.5,d=4 me=0.5,d=4\,\mathrm m: one ellipse with vertices x=−4 and 43 mx=-4\text{ and }\frac43\,\mathrm m; the distance ratio is 0.5.
  • Changed e=1,d=4 me=1,d=4\,\mathrm m: a left-opening parabola with vertex x=2x=2 m\mathrm m and no right positive-radius branch.

Boundary and failure behavior

Boundary: e=0e=0 is rejected. At either hyperbolic angular limit the radius diverges, so no finite plotted point exists. At e=1e=1 only the left positive-radius branch remains.
