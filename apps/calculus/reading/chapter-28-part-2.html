  4. Problem 4. Interpretation Why must a step study reach the same final time in every run?

    Hint. Separate time evolution from numerical error.

    Solution. A refinement study compares approximations to the same exact state y(T)y(T). If runs end at different times TiT_i, their differences include real evolution as well as numerical error. Choose grids that reach a common TT, or interpolate consistently.

  5. Problem 5. Interpretation If xx and vv form a state, may Euler update vv using the just-updated xx?

    Hint. Check which state supplies the stage slope.

    Solution. Standard forward Euler evaluates every component's derivative from the old state. Updating vv with the new xx creates a different sequential scheme and can change its accuracy or stability.

  8. Problem 8. Interpretation How many ff evaluations per fixed step do Euler, Heun, and classical RK4 use?

    Solution. Euler evaluates ff once per step; explicit Heun evaluates an initial and predicted-endpoint slope, twice; classical RK4 evaluates four slopes. These fixed-step counts exclude adaptive retries and special reuse.

    const SCALAR_IVP_VERSION = "scalar-ivp-reference-v1";
const MAX_STEPS = 10000;
const MAX_ABS_Y = 1e6;

function solveScalarIVP({ f, inDomain, method, t0, t1, y0, N }) {
  const methods = new Set(["Euler", "Heun", "RK4"]);
  if (typeof f !== "function" || typeof inDomain !== "function") {
    throw new TypeError("Supply pure terminating f(t,y) and inDomain(t,y) functions.");
  }
  if (!methods.has(method)) {
    throw new RangeError("Method must be Euler, Heun or RK4.");
  }
  if (!Number.isInteger(N) || N < 1 || N > MAX_STEPS) {
    throw new RangeError("N must be an integer from 1 to 10000.");
  }
  if (!Number.isFinite(t0) || !Number.isFinite(t1) || !(t0 < t1)) {
    throw new RangeError("Require finite t0 < t1.");
  }

  function finite(value, label) {
    if (!Number.isFinite(value)) {
      throw new RangeError(label + " must be finite.");
    }
    return value;
  }

  const span = finite(t1 - t0, "Interval length");
  const nominalH = finite(span / N, "Nominal step");
  if (!(nominalH > 0)) {
    throw new RangeError("The nominal step must be positive.");
  }

  function state(t, y) {
    finite(t, "Stage time");
    finite(y, "Stage state");
    if (t < t0 || t > t1) {
      throw new RangeError("Stage time lies outside the integration interval.");
    }
    if (Math.abs(y) > MAX_ABS_Y) {
      throw new RangeError("State exceeds the declared magnitude cap.");
    }
    if (inDomain(t, y) !== true) {
      throw new RangeError("State lies outside the declared domain.");
    }
    return y;
  }

  let evaluations = 0;
  function rate(t, y) {
    state(t, y);
    evaluations += 1;
    return finite(f(t, y), "Rate");
  }

  let y = state(t0, y0);
  const points = [[t0, y]];

  for (let i = 0; i < N; i += 1) {
    const t = points[i][0];
    const nextT = i + 1 === N
      ? t1
      : finite(t0 + span * ((i + 1) / N), "Grid time");
    const h = finite(nextT - t, "Represented panel width");
    if (!(h > 0)) {
      throw new RangeError("Time panels must remain distinguishable.");
    }

    const k1 = rate(t, y);
    let increment;

    if (method === "Euler") {
      increment = finite(h * k1, "Euler increment");
    } else if (method === "Heun") {
      const predictor = state(
        nextT,
        finite(y + finite(h * k1, "Predictor increment"), "Predictor")
      );
      const k2 = rate(nextT, predictor);
      const slopeSum = finite(k1 + k2, "Heun slope sum");
      increment = finite(h * slopeSum / 2, "Heun increment");
    } else {
      const middleT = finite(t + h / 2, "Midpoint time");
      if (!(t < middleT && middleT < nextT)) {
        throw new RangeError("RK4 midpoint is not representable inside its panel.");
      }
      const y2 = state(
        middleT,
        finite(y + finite(h * k1 / 2, "RK4 stage 2 increment"), "RK4 stage 2")
      );
      const k2 = rate(middleT, y2);
      const y3 = state(
        middleT,
        finite(y + finite(h * k2 / 2, "RK4 stage 3 increment"), "RK4 stage 3")
      );
      const k3 = rate(middleT, y3);
      const y4 = state(
        nextT,
        finite(y + finite(h * k3, "RK4 stage 4 increment"), "RK4 stage 4")
      );
      const k4 = rate(nextT, y4);
      const weighted = finite(
        k1 + 2 * k2 + 2 * k3 + k4,
        "RK4 weighted slope sum"
      );
      increment = finite(h * weighted / 6, "RK4 increment");
    }

    y = state(nextT, finite(y + increment, "Updated state"));
    points.push([nextT, y]);
  }

  return {
    version: SCALAR_IVP_VERSION,
    method,
    N,
    nominalH,
    evaluations,
    points
  };
}

Chapter synthesis

  • A solver approximates rate accumulation; its order requires smoothness and controls error scaling on fixed intervals.
  • Stability, positivity, accuracy and invariant preservation are separate checks.
  • A reproducible capstone joins a justified model, explicit numerical method, refinement evidence and a conclusion with stated limits.

Common mistakes to check

  • “A smooth plot proves the answer is accurate.” Sampling and line drawing can hide substantial error; compare steps, special cases and invariants. Test with problem 29
  • “Fourth order means four correct decimal places.” Under the usual smoothness and stability assumptions on a fixed interval, fourth order means global error O(h4)O(h^4) as h→0h\to0, not four correct decimal places. Test with problem 7
